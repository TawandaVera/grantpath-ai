### **B. Actionability Test (Every Grant Must Pass)**

```python
def actionability_test(grant):
    """
    Returns: PASS, NEEDS_VERIFICATION, or REJECTED
    ALL six criteria must be satisfied for PASS.
    """
    
    criteria = {
        "live_opportunity": is_currently_open(grant),
        "official_url": is_funder_controlled_or_official_portal(grant.url),
        "direct_application_link": url_leads_to_application_endpoint(grant.url),
        "deadline_present": has_deadline_or_rolling(grant),
        "applicant_type_identifiable": can_identify_applicant_type(grant),
        "not_informational_only": is_not_historical_or_informational(grant)
    }
    
    if all(criteria.values()):
        return "PASS"
    elif any(criteria.values()):
        return "NEEDS_VERIFICATION", [k for k, v in criteria.items() if not v]
    else:
        return "REJECTED — NON-ACTIONABLE LINK"
```

### **C. URL Validation Engine**

```python
ALLOWED_URL_TYPES = [
    "application_portal",        # Direct login/apply page
    "application_form",          # Downloadable or online form
    "application_package",       # Official NOFO/RFP/package
    "official_opportunity_page", # Contains Apply/Start Application button
    "instructions_pdf"           # Official submission entry point
]

DISALLOWED_URL_TYPES = [
    "press_release",
    "award_announcement",
    "funding_news_roundup",
    "search_results_page",       # Unless it IS the official opportunity page
    "generic_homepage",
    "listicle",
    "third_party_directory",
    "additional_opportunities_page",
    "past_award_recipient_page",
    "archived_opportunity_page", # Unless explicitly marked as archive
    "broad_agency_landing_page"  # When specific opportunity page exists
]

def validate_url(url, grant_context):
    url_type = classify_url(url)
    
    if url_type in DISALLOWED_URL_TYPES:
        return {
            "valid": False,
            "reason": f"REJECTED — NON-ACTIONABLE LINK: {url_type}",
            "recommendation": "Search for direct application endpoint"
        }
    
    if url_type not in ALLOWED_URL_TYPES:
        return {
            "valid": False,
            "reason": "URL does not match any allowed application endpoint type",
            "max_confidence": "Low"
        }
    
    return {
        "valid": True,
        "url_type": url_type,
        "confidence": determine_confidence(url, grant_context)
    }
```

## **5. DATA SCHEMA**

```sql
-- Master pipeline tracking
CREATE TABLE CoPilotPipeline (
    pipeline_id UUID PRIMARY KEY,
    session_id UUID,
    created_at TIMESTAMP,
    criteria JSON,
    master_narrative_hash CHAR(64),
    current_stage VARCHAR(20),
    total_discovered INT,
    total_validated INT,
    total_rejected INT,
    total_selected INT,
    total_packs_complete INT,
    status ENUM('active', 'paused', 'complete', 'abandoned')
);

-- HIL checkpoint log (immutable audit trail)
CREATE TABLE HILCheckpoints (
    checkpoint_id UUID PRIMARY KEY,
    pipeline_id UUID REFERENCES CoPilotPipeline(pipeline_id),
    grant_fingerprint CHAR(32),
    stage VARCHAR(50),
    section VARCHAR(100),
    presented_at TIMESTAMP,
    decision ENUM('approve', 'edit', 'regenerate', 'verify', 'pause', 'block'),
    decision_at TIMESTAMP,
    decision_by VARCHAR(100),
    edit_instructions TEXT,
    notes TEXT,
    time_to_decision_seconds INT,
    
    INDEX idx_pipeline_stage (pipeline_id, stage),
    INDEX idx_pending (decision_at)  -- NULL = pending
);

-- URL validation log
CREATE TABLE URLValidationLog (
    validation_id UUID PRIMARY KEY,
    grant_fingerprint CHAR(32),
    url VARCHAR(500),
    url_type VARCHAR(50),
    validation_timestamp TIMESTAMP,
    is_valid BOOLEAN,
    rejection_reason TEXT,
    confidence ENUM('high', 'medium', 'low'),
    source_priority_level INT,
    http_status_code INT,
    redirect_chain JSON,
    final_destination_url VARCHAR(500)
);

-- Application pack versions
CREATE TABLE ApplicationPacks (
    pack_id UUID PRIMARY KEY,
    pipeline_id UUID,
    grant_fingerprint CHAR(32),
    version INT,
    created_at TIMESTAMP,
    
    -- Pack sections stored as JSON
    opportunity_summary JSON,
    eligibility_readiness JSON,
    response_matrix JSON,
    draft_narratives JSON,     -- Array of {question, draft, sources, confidence, flags}
    attachment_checklist JSON,
    missing_info_checklist JSON,
    verification_log JSON,
    hil_review_log JSON,
    
    -- Quality metrics
    total_sections INT,
    sections_approved INT,
    sections_pending INT,
    sections_flagged INT,
    needs_verification_count INT,
    needs_info_count INT,
    overall_confidence ENUM('high', 'medium', 'low'),
    
    -- Status
    pack_status ENUM('drafting', 'in_review', 'approved', 'submitted'),
    
    INDEX idx_status (pack_status, created_at)
);
```
