```python
# Priority Ranking Algorithm
def select_top_grants(grants, class_id):
    # Filter: Only GO and PREP states
    eligible = [g for g in grants if g.state in ["GO", "PREP"] and g.class_id == class_id]
    
    # Sort: Primary by match_score DESC, secondary by deadline ASC
    sorted_grants = sorted(eligible, key=lambda x: (-x.match_score, x.deadline))
    
    # Return top 20 per class
    return sorted_grants[:20]
```

## **4. DATA SCHEMA (Packages Table)**

```sql
CREATE TABLE GrantPackages (
    package_id UUID PRIMARY KEY,
    grant_fingerprint CHAR(32) REFERENCES JobsRaw(grant_fingerprint),
    assessment_id UUID REFERENCES GrantAssessments(assessment_id),
    packaging_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Package Status
    package_status ENUM('draft', 'ready', 'submitted', 'archived'),
    completion_percentage INT,
    
    -- Dossier Components (stored as JSON for flexibility)
    dossier_summary JSON,
    eligibility_checklist JSON,
    required_forms JSON,
    portal_text JSON,
    
    -- Progress Tracking
    checklist_completed_at TIMESTAMP,
    forms_downloaded_at TIMESTAMP,
    text_approved_at TIMESTAMP,
    submitted_at TIMESTAMP,
    
    -- Assignments
    assigned_to VARCHAR(100),
    writer_assigned VARCHAR(100),
    reviewer_assigned VARCHAR(100),
    
    -- Version Control
    dossier_version INT DEFAULT 1,
    last_modified_by VARCHAR(100),
    last_modified_at TIMESTAMP,
    
    -- Audit
    packaging_agent_version VARCHAR(20),
    notes TEXT,
    
    INDEX idx_package_status (package_status),
    INDEX idx_deadline_package (deadline, package_status),
    INDEX idx_assigned (assigned_to, package_status)
);
```
